cmake_minimum_required(VERSION 3.19)
project(clean_vpn_native LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(CVPN_SANITIZE "Address/undefined sanitizers for native engine and tests" OFF)
option(CVPN_LOW_MEMORY_BUILD "Omit compiler debug information without reducing optimization (not a RAM guarantee)" OFF)
if(CVPN_LOW_MEMORY_BUILD)
  if(NOT CMAKE_C_COMPILER_ID MATCHES "GNU|Clang" OR NOT CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    message(FATAL_ERROR "CVPN_LOW_MEMORY_BUILD requires GCC or Clang")
  endif()
  # Inherited by BoringSSL/nghttp2 as well as our targets, AFTER build-type
  # flags. Toggling OFF removes this option, without sticky cached flag edits.
  add_compile_options("$<$<COMPILE_LANGUAGE:C,CXX>:-g0>")
endif()
set(CVPN_BORINGSSL_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../boring_tls/build/_deps/boringssl-src" CACHE PATH "Existing pinned BoringSSL source")
set(CVPN_JSON_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../boring_tls/build/_deps/json-src" CACHE PATH "Existing nlohmann/json source")
find_package(Git REQUIRED)
find_package(Threads REQUIRED)
execute_process(COMMAND ${GIT_EXECUTABLE} -C ${CVPN_BORINGSSL_SOURCE} rev-parse HEAD OUTPUT_VARIABLE BSSL_HEAD OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
if(NOT BSSL_HEAD STREQUAL "a7481f34712bc056a47ab91015536166b3a6cebb")
  message(FATAL_ERROR "Use the repository's pinned BoringSSL source; build the existing helper first")
endif()
if(NOT EXISTS "${CVPN_JSON_SOURCE}/single_include/nlohmann/json.hpp")
  message(FATAL_ERROR "Missing JSON headers; build the existing helper first")
endif()
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
add_subdirectory(${CVPN_BORINGSSL_SOURCE} bssl EXCLUDE_FROM_ALL)
include(FetchContent)
set(BROTLI_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
FetchContent_Declare(brotli GIT_REPOSITORY https://github.com/google/brotli.git GIT_TAG 028fb5a23661f123017c060daa546b55cf4bde29)
FetchContent_MakeAvailable(brotli)
set(ENABLE_LIB_ONLY ON CACHE BOOL "" FORCE)
set(BUILD_STATIC_LIBS ON CACHE BOOL "" FORCE)
FetchContent_Declare(nghttp2 GIT_REPOSITORY https://github.com/nghttp2/nghttp2.git GIT_TAG 534b74b72524e962c18c7146470914632ca7eb2d)
FetchContent_MakeAvailable(nghttp2)
# Private ALPS adapter must be audited again when nghttp2 is upgraded.
execute_process(COMMAND ${GIT_EXECUTABLE} -C ${nghttp2_SOURCE_DIR} rev-parse HEAD OUTPUT_VARIABLE NGHTTP2_HEAD OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
if(NOT NGHTTP2_HEAD STREQUAL "534b74b72524e962c18c7146470914632ca7eb2d")
  message(FATAL_ERROR "ALPS adapter requires the audited nghttp2 revision")
endif()
add_library(cvpn_http2_alps STATIC http2_alps_adapter.c)
target_include_directories(cvpn_http2_alps PRIVATE ${nghttp2_SOURCE_DIR}/lib ${nghttp2_BINARY_DIR} PUBLIC ${nghttp2_SOURCE_DIR}/lib/includes ${nghttp2_BINARY_DIR}/lib/includes)
target_link_libraries(cvpn_http2_alps PUBLIC nghttp2_static)
target_compile_options(cvpn_http2_alps PRIVATE -Wall -Wextra -Werror)
add_executable(http2-alps-test http2_alps_test.cc)
target_include_directories(http2-alps-test PRIVATE ${CVPN_BORINGSSL_SOURCE}/include)
target_link_libraries(http2-alps-test PRIVATE cvpn_http2_alps ssl crypto)
target_compile_options(http2-alps-test PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(clean-vpn-engine engine.cc)
add_executable(clean-vpn-engine-fixture engine.cc)
target_compile_definitions(clean-vpn-engine-fixture PRIVATE CVPN_TEST_PACKET_FD=1)
target_link_libraries(clean-vpn-engine-fixture PRIVATE brotlienc)
foreach(engine clean-vpn-engine clean-vpn-engine-fixture)
target_include_directories(${engine} PRIVATE ${CVPN_BORINGSSL_SOURCE}/include ${CVPN_JSON_SOURCE}/single_include ${nghttp2_SOURCE_DIR}/lib/includes ${nghttp2_BINARY_DIR}/lib/includes)
target_link_libraries(${engine} PRIVATE ssl crypto cvpn_http2_alps brotlidec Threads::Threads)
target_compile_options(${engine} PRIVATE -Wall -Wextra -Wpedantic -Werror)
endforeach()
add_executable(protocol-test protocol_test.cc)
add_executable(transparent-test transparent_test.cc)
add_executable(transparent-socket-test transparent_socket_test.cc)
add_executable(transparent-replay-test transparent_replay_test.cc)
add_executable(combo-test combo_test.cc)
add_executable(throughput-test throughput_test.cc)
add_executable(tls-profile-test tls_profile_test.cc)
add_executable(profile-credentials-fixture profile_credentials_fixture.cc)
target_include_directories(profile-credentials-fixture PRIVATE ${CVPN_BORINGSSL_SOURCE}/include)
target_link_libraries(profile-credentials-fixture PRIVATE crypto)
target_compile_options(profile-credentials-fixture PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(certificate-compression-test certificate_compression_test.cc)
target_include_directories(certificate-compression-test PRIVATE ${CVPN_BORINGSSL_SOURCE}/include)
target_link_libraries(certificate-compression-test PRIVATE ssl crypto brotlidec brotlienc)
target_compile_options(certificate-compression-test PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_link_libraries(tls-profile-test PRIVATE brotlidec)
target_compile_definitions(transparent-replay-test PRIVATE CVPN_TEST_REPLAY_JOURNAL=1)
foreach(target transparent-test transparent-socket-test transparent-replay-test combo-test throughput-test tls-profile-test)
target_include_directories(${target} PRIVATE ${CVPN_BORINGSSL_SOURCE}/include ${CVPN_JSON_SOURCE}/single_include)
target_link_libraries(${target} PRIVATE ssl crypto Threads::Threads)
target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
endforeach()
add_executable(service-notify-test service_notify_test.cc)
target_compile_options(service-notify-test PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(dns-wire-test dns_wire_test.cc)
add_executable(dns-relay-test dns_relay_test.cc)
target_include_directories(dns-relay-test PRIVATE ${CVPN_BORINGSSL_SOURCE}/include)
target_link_libraries(dns-relay-test PRIVATE crypto Threads::Threads)
target_compile_options(dns-relay-test PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_compile_options(dns-wire-test PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_compile_options(protocol-test PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(integration-test integration_test.cc)
add_executable(socket-test socket_test.cc)
target_compile_options(socket-test PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_include_directories(integration-test PRIVATE ${CVPN_JSON_SOURCE}/single_include)
target_compile_options(integration-test PRIVATE -Wall -Wextra -Wpedantic -Werror)
if(CVPN_SANITIZE)
  foreach(target clean-vpn-engine clean-vpn-engine-fixture protocol-test dns-wire-test dns-relay-test integration-test service-notify-test transparent-test transparent-socket-test transparent-replay-test combo-test throughput-test tls-profile-test certificate-compression-test http2-alps-test cvpn_http2_alps brotlidec brotlicommon)
    target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(${target} PRIVATE -fsanitize=address,undefined)
  endforeach()
endif()
enable_testing()
add_test(NAME http2-alps COMMAND http2-alps-test)
add_test(NAME protocol COMMAND protocol-test)
add_test(NAME dns-wire COMMAND dns-wire-test)
add_test(NAME service-notify COMMAND service-notify-test)
add_test(NAME transparent-codec COMMAND transparent-test)
add_test(NAME transparent-replay COMMAND transparent-replay-test)
add_test(NAME combo COMMAND combo-test)
add_test(NAME throughput-wire COMMAND throughput-test --self-test)
add_test(NAME tls-profile COMMAND tls-profile-test ${CMAKE_CURRENT_SOURCE_DIR}/../../fixtures/native-tls-profiles/cold-h2.json)
add_test(NAME certificate-compression COMMAND certificate-compression-test)
